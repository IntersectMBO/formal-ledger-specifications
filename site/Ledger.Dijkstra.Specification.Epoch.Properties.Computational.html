module _ {eps : EpochState} {e : Epoch} where

  EPOCH-total : ∃[ eps' ] _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps'
  EPOCH-total =
    -, EPOCH
         ( POOLREAP-total  .proj₂
         , SNAP-total      .proj₂
         , RATIFIES-total' .proj₂)

  EPOCH-deterministic : ∀ eps' eps''
                      → _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps'
                      → _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps''
                      → eps' ≡ eps''
  EPOCH-deterministic
      eps'
      eps''
      (EPOCH
        {dState' = dState'₁}
        {acnt' = acnt'₁}
        {pState'' = pState'₁}
        {ss' = ss'₁}
        (p₁ , p₂ , p₃)
      )
      (EPOCH
        {dState' = dState'₂}
        {acnt' = acnt'₂}
        {pState'' = pState'₂}
        {ss' = ss'₂}
        (p₁' , p₂' , p₃')
      ) = eps'≡eps''
       where
         ls : LedgerState
         ls = LedgerStateOf eps

         es : EnactState
         es = EnactStateOf (RatifyStateOf eps)

         govUpd : Governance-Update
         govUpd = GovernanceUpdate.updates ls (RatifyStateOf eps)

         govSt' = Governance-Update.govSt' govUpd

         module pPRUpd =  Pre-POOLREAP-Update (Pre-POOLREAPUpdate.updates e ls es govUpd)

         pPRUpd₁ = Post-POOLREAPUpdate.updates es ls dState'₁ acnt'₁ govUpd
         module pPRUpd₁ = Post-POOLREAP-Update pPRUpd₁

         pPRUpd₂ = Post-POOLREAPUpdate.updates es ls dState'₂ acnt'₂ govUpd
         module pPRUpd₂ = Post-POOLREAP-Update pPRUpd₂

         prs'≡prs'' : $\begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1740}{\htmlId{2673}{\htmlClass{Bound}{\text{acnt'₁}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1713}{\htmlId{2682}{\htmlClass{Bound}{\text{dState'₁}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1768}{\htmlId{2693}{\htmlClass{Bound}{\text{pState'₁}}}}\, \end{pmatrix}$ ≡
                      $\begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1889}{\htmlId{2731}{\htmlClass{Bound}{\text{acnt'₂}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1862}{\htmlId{2740}{\htmlClass{Bound}{\text{dState'₂}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1917}{\htmlId{2751}{\htmlClass{Bound}{\text{pState'₂}}}}\, \end{pmatrix}$
         prs'≡prs'' = POOLREAP-deterministic-≡ refl refl p₁ p₁'

         pPRUpd₁≡pPRUpd₂ : pPRUpd₁ ≡ pPRUpd₂
         pPRUpd₁≡pPRUpd₂ rewrite (cong PoolReapState.dState prs'≡prs'') | (cong PoolReapState.acnt prs'≡prs'') = refl

         ls'₁ = $\begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.html#21489}{\htmlId{3010}{\htmlClass{Function}{\text{pPRUpd.utxoSt'}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#2252}{\htmlId{3027}{\htmlClass{Function}{\text{govSt'}}}}\, \\ \begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.html#22435}{\htmlId{3038}{\htmlClass{Function}{\text{pPRUpd₁.dState''}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1768}{\htmlId{3057}{\htmlClass{Bound}{\text{pState'₁}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.html#21468}{\htmlId{3068}{\htmlClass{Function}{\text{pPRUpd.gState'}}}}\, \end{pmatrix} \end{pmatrix}$
         ls'₂ = $\begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.html#21489}{\htmlId{3107}{\htmlClass{Function}{\text{pPRUpd.utxoSt'}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#2252}{\htmlId{3124}{\htmlClass{Function}{\text{govSt'}}}}\, \\ \begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.html#22435}{\htmlId{3135}{\htmlClass{Function}{\text{pPRUpd₂.dState''}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1917}{\htmlId{3154}{\htmlClass{Bound}{\text{pState'₂}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.html#21468}{\htmlId{3165}{\htmlClass{Function}{\text{pPRUpd.gState'}}}}\, \end{pmatrix} \end{pmatrix}$

         ls'₁≡ls'₂ : ls'₁ ≡ ls'₂
         ls'₁≡ls'₂ = cong₂ (λ ds ps → $\begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.html#21489}{\htmlId{3260}{\htmlClass{Function}{\text{pPRUpd.utxoSt'}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#2252}{\htmlId{3277}{\htmlClass{Function}{\text{govSt'}}}}\, \\ \begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3250}{\htmlId{3288}{\htmlClass{Bound}{\text{ds}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3253}{\htmlId{3293}{\htmlClass{Bound}{\text{ps}}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.html#21468}{\htmlId{3298}{\htmlClass{Function}{\text{pPRUpd.gState'}}}}\, \end{pmatrix} \end{pmatrix}$)
                       (cong Post-POOLREAP-Update.dState'' pPRUpd₁≡pPRUpd₂)
                       (cong PoolReapState.pState prs'≡prs'')

         ss'₁≡ss'₂ : ss'₁ ≡ ss'₂
         ss'₁≡ss'₂ = SNAP-deterministic-≡ ls'₁≡ls'₂ p₂ p₂'

         mkΓ : Snapshots → Post-POOLREAP-Update → LedgerState → RatifyEnv
         mkΓ ss'' pPR ls'' =
           $\begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.html#16162}{\htmlId{3668}{\htmlClass{Function}{\text{calculateVDelegDelegatedStake}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1258}{\htmlId{3698}{\htmlClass{Bound}{\text{e}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.html#21489}{\htmlId{3700}{\htmlClass{Function}{\text{pPRUpd.utxoSt'}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#2252}{\htmlId{3715}{\htmlClass{Function}{\text{govSt'}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.html#21468}{\htmlId{3722}{\htmlClass{Function}{\text{pPRUpd.gState'}}}}\, \,\htmlId{3737}{\htmlClass{Symbol}{\text{(}}}\,\,\href{Ledger.Dijkstra.Specification.Epoch.html#22435}{\htmlId{3738}{\htmlClass{Field}{\text{Post-POOLREAP-Update.dState''}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3644}{\htmlId{3768}{\htmlClass{Bound}{\text{pPR}}}}\,\,\htmlId{3771}{\htmlClass{Symbol}{\text{)}}}\,
           \\ \,\href{Ledger.Dijkstra.Specification.Epoch.html#16388}{\htmlId{3786}{\htmlClass{Function}{\text{calculatePoolDelegatedStakeForVoting}}}}\, \,\htmlId{3823}{\htmlClass{Symbol}{\text{(}}}\,\,\href{Ledger.Dijkstra.Specification.Rewards.html#7972}{\htmlId{3824}{\htmlClass{Field}{\text{Snapshots.mark}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3639}{\htmlId{3839}{\htmlClass{Bound}{\text{ss''}}}}\,\,\htmlId{3843}{\htmlClass{Symbol}{\text{)}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#2252}{\htmlId{3845}{\htmlClass{Function}{\text{govSt'}}}}\,
           \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#1258}{\htmlId{3865}{\htmlClass{Bound}{\text{e}}}}\,
           \\ \,\href{Ledger.Dijkstra.Specification.Gov.Actions.html#5444}{\htmlId{3880}{\htmlClass{Field}{\text{DRepsOf}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3648}{\htmlId{3888}{\htmlClass{Bound}{\text{ls''}}}}\,
           \\ \,\href{Ledger.Dijkstra.Specification.Certs.html#5157}{\htmlId{3906}{\htmlClass{Field}{\text{CCHotKeysOf}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3648}{\htmlId{3918}{\htmlClass{Bound}{\text{ls''}}}}\,
           \\ \,\href{Ledger.Prelude.Base.html#1023}{\htmlId{3936}{\htmlClass{Field}{\text{TreasuryOf}}}}\, \,\htmlId{3947}{\htmlClass{Symbol}{\text{(}}}\,\,\href{Ledger.Dijkstra.Specification.Epoch.html#22463}{\htmlId{3948}{\htmlClass{Field}{\text{Post-POOLREAP-Update.acnt''}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3644}{\htmlId{3976}{\htmlClass{Bound}{\text{pPR}}}}\,\,\htmlId{3979}{\htmlClass{Symbol}{\text{)}}}\,
           \\ \,\href{Ledger.Dijkstra.Specification.Certs.html#5414}{\htmlId{3994}{\htmlClass{Field}{\text{PoolsOf}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3648}{\htmlId{4002}{\htmlClass{Bound}{\text{ls''}}}}\,
           \\ \,\href{Ledger.Dijkstra.Specification.Gov.Actions.html#4773}{\htmlId{4020}{\htmlClass{Field}{\text{VoteDelegsOf}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#3648}{\htmlId{4033}{\htmlClass{Bound}{\text{ls''}}}}\, \end{pmatrix}$

         Γ≡Γ' : mkΓ ss'₁ pPRUpd₁ ls'₁ ≡ mkΓ ss'₂ pPRUpd₂ ls'₂
         Γ≡Γ' = trans (cong (λ x → mkΓ x pPRUpd₁ ls'₁) ss'₁≡ss'₂)
                (trans (cong (λ x → mkΓ ss'₂ x ls'₁) pPRUpd₁≡pPRUpd₂)
                       (cong (mkΓ ss'₂ pPRUpd₂) ls'₁≡ls'₂))

         fut'≡fut'' : RatifyStateOf eps' ≡ RatifyStateOf eps''
         fut'≡fut'' = RATIFIES-deterministic-≡ Γ≡Γ' refl refl p₃ p₃'

         eps'≡eps'' : eps' ≡ eps''
         eps'≡eps''
           rewrite ss'₁≡ss'₂
             | cong PoolReapState.pState prs'≡prs''
             | cong Post-POOLREAP-Update.acnt'' pPRUpd₁≡pPRUpd₂
             | cong Post-POOLREAP-Update.dState'' pPRUpd₁≡pPRUpd₂
             | fut'≡fut'' = refl

  EPOCH-complete : ∀ eps' → _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps' → proj₁ EPOCH-total ≡ eps'
  EPOCH-complete eps' p = EPOCH-deterministic (proj₁ EPOCH-total) eps' (proj₂ EPOCH-total) p

  abstract
    EPOCH-total' : ∃[ eps' ] _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps'
    EPOCH-total' = EPOCH-total

    EPOCH-complete' : ∀ eps' → _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps' → proj₁ EPOCH-total' ≡ eps'
    EPOCH-complete' = EPOCH-complete

instance
  Computational-EPOCH : Computational _⊢_⇀⦇_,EPOCH⦈_ ⊥
  Computational-EPOCH .computeProof Γ s sig = success EPOCH-total'
  Computational-EPOCH .completeness Γ s sig s' h = cong success (EPOCH-complete' s' h)

module _ {e : Epoch} where

  NEWEPOCH-total : ∀ nes'' → ∃[ nes' ] _ ⊢ nes'' ⇀⦇ e ,NEWEPOCH⦈ nes'
  NEWEPOCH-total nes with e ≟ NewEpochState.lastEpoch nes + 1 | NewEpochState.ru nes | inspect NewEpochState.ru nes
  ... | yes p | just ru | PE.[ refl ] =  $\begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#5358}{\htmlId{5605}{\htmlClass{Bound}{\text{e}}}}\, \\ \,\htmlId{5609}{\htmlClass{Symbol}{\text{\_}}}\, \\ \,\htmlId{5613}{\htmlClass{Symbol}{\text{\_}}}\, \\ \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#4923}{\htmlId{5617}{\htmlClass{Function}{\text{EPOCH-total'}}}}\, \,\htmlId{5630}{\htmlClass{Symbol}{\text{.}}}\,\,\href{Data.Product.Base.html#636}{\htmlId{5631}{\htmlClass{Field}{\text{proj₁}}}}\, \\ \,\href{Agda.Builtin.Maybe.html#194}{\htmlId{5639}{\htmlClass{InductiveConstructor}{\text{nothing}}}}\, \\ \,\htmlId{5649}{\htmlClass{Symbol}{\text{\_}}}\, \end{pmatrix}$
                                      , NEWEPOCH-New (p , EPOCH-total' .proj₂)
  ... | yes p | nothing | PE.[ refl ] = $\begin{pmatrix} \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#5358}{\htmlId{5774}{\htmlClass{Bound}{\text{e}}}}\, \\ \,\htmlId{5778}{\htmlClass{Symbol}{\text{\_}}}\, \\ \,\htmlId{5782}{\htmlClass{Symbol}{\text{\_}}}\, \\ \,\href{Data.Product.Base.html#636}{\htmlId{5786}{\htmlClass{Field}{\text{proj₁}}}}\, \,\href{Ledger.Dijkstra.Specification.Epoch.Properties.Computational.html#4923}{\htmlId{5792}{\htmlClass{Function}{\text{EPOCH-total'}}}}\, \\ \,\href{Agda.Builtin.Maybe.html#194}{\htmlId{5807}{\htmlClass{InductiveConstructor}{\text{nothing}}}}\, \\ \,\htmlId{5817}{\htmlClass{Symbol}{\text{\_}}}\, \end{pmatrix}$
                                      , NEWEPOCH-No-Reward-Update (p , EPOCH-total' .proj₂)
  ... | no ¬p | _ | _ = -, NEWEPOCH-Not-New ¬p

  NEWEPOCH-complete : ∀ nes nes' → _ ⊢ nes ⇀⦇ e ,NEWEPOCH⦈ nes' → proj₁ (NEWEPOCH-total nes) ≡ nes'
  NEWEPOCH-complete nes nes' h with e ≟ NewEpochState.lastEpoch nes + 1 | NewEpochState.ru nes | inspect NewEpochState.ru nes | h
  ... | yes p | just ru | PE.[ refl ] | NEWEPOCH-New (x , x₁) rewrite EPOCH-complete' _ x₁ = refl
  ... | yes p | ru | PE.[ refl ] | NEWEPOCH-Not-New x = ⊥-elim $ x p
  ... | yes p | nothing | PE.[ refl ] | NEWEPOCH-No-Reward-Update (x , x₁) rewrite EPOCH-complete' _ x₁ = refl
  ... | no ¬p | ru | PE.[ refl ] | NEWEPOCH-New (x , x₁)  = ⊥-elim $ ¬p x
  ... | no ¬p | ru | PE.[ refl ] | NEWEPOCH-Not-New x = refl
  ... | no ¬p | nothing | PE.[ refl ] | NEWEPOCH-No-Reward-Update (x , x₁) = ⊥-elim $ ¬p x

instance
  Computational-NEWEPOCH : Computational _⊢_⇀⦇_,NEWEPOCH⦈_ ⊥
  Computational-NEWEPOCH .computeProof Γ s sig = success (NEWEPOCH-total _)
  Computational-NEWEPOCH .completeness Γ s sig s' h = cong success (NEWEPOCH-complete _ s' h)
