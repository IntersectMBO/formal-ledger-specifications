-- The decisions of the certificate branch, one per premise of CERTIFY-EB.
pending? : (m : Maybe LastAppliedBlock) → Dec (∃[ la ] m ≡ just la)
pending? nothing   = no λ where (_ , ())
pending? (just la) = yes (la , refl)

blockSlot : Block → Slot
blockSlot b = BHBody.slot (BHeader.bhbody (Block.bheader b))

pendingFailure : LastAppliedBlock → String
pendingFailure la with LastAppliedBlock.announcedEB la
... | nothing = "the last applied block announced no EB"
... | just _  = "the certified EB is not the one the last applied block announced"

module _ (Γ : CertifyEnv) (la : LastAppliedBlock) (cert : EBCert) (reb : ResolvedEB) where
  open CertifyEnv Γ; open ResolvedEB reb
  open LastAppliedBlock la using (headerHash) renaming (slot to announcingSlot)
  open PParams (PParamsOf enactState) using (leiosQuorumStakeThreshold)

  pendingEB? : Dec (pendingEB la ≡ just (hashEB eb))
  pendingEB? = pendingEB la ≟ just (hashEB eb)

  delay? : Dec (announcingSlot + certificationDelay (PParamsOf enactState) ≤ slot)
  delay? = ¿ _ ¿

  sameEpoch? : Dec (epoch announcingSlot ≡ epoch slot)
  sameEpoch? = epoch announcingSlot ≟ epoch slot

  validCert? : Dec (ValidEBCert committee leiosQuorumStakeThreshold (rbHeaderHashBytes headerHash) cert)
  validCert? = ¿ _ ¿

  -- The EB step's checks besides LEDGERS, against the announcing block's parameters.
  ebChecks? : Dec ( EBWellFormed eb
                  × EndorserBlock.ebTxRefs eb ≡ map (λ tx → txRefHash tx , SizeOf tx) closure
                  × ebSize eb ≤ PParams.leiosMaxEBSize (PParamsOf enactState)
                  × ∑ˡ[ tx ← closure ] SizeOf tx ≤ PParams.leiosMaxEBTxsSize (PParamsOf enactState) )
  ebChecks? = wellFormed? ×-dec ¿ _ ¿
    where
      refs = EndorserBlock.ebTxRefs eb
      nonempty? : Dec (refs ≢ [])
      nonempty? with refs
      ... | []    = no λ p → p refl
      ... | _ ∷ _ = yes λ ()
      wellFormed? : Dec (EBWellFormed eb)
      wellFormed? = map′ (λ (n , u) → ebWellFormed (n , u)) (λ where (ebWellFormed (n , u)) → n , u)
                         (nonempty? ×-dec unique? _≟_ (map proj₁ refs))

instance

      Computational-CERTIFY : Computational _⊢_⇀⦇_,CERTIFY⦈_ String
  Computational-CERTIFY .computeProof Γ ls (nothing , nothing) = success (ls , CERTIFY-None)
  Computational-CERTIFY .computeProof Γ ls (just _ , nothing) =
    failure "the block carries an EB certificate but no resolved EB came beside it"
  Computational-CERTIFY .computeProof Γ ls (nothing , just _) =
    failure "a resolved EB came beside a block that carries no EB certificate"
  Computational-CERTIFY .computeProof Γ ls (just cert , just reb)
    with pending? (CertifyEnv.lastApplied Γ)
  ... | no _ = failure "the chain has no last applied block, so no EB is pending"
  ... | yes (la , la≡)
    with pendingEB? Γ la cert reb
  ... | no _ = failure (pendingFailure la)
  ... | yes pend
    with delay? Γ la cert reb
  ... | no _ = failure "the certificate comes before the certification delay has elapsed"
  ... | yes d
    with sameEpoch? Γ la cert reb
  ... | no _ = failure "the certificate crosses an epoch boundary, so the announcement has been retired"
  ... | yes e
    with validCert? Γ la cert reb
  ... | no _ = failure "the EB certificate is not valid against the announcing epoch's committee"
  ... | yes vc = do
    ls₁ , ext ← computeProof _ ls (ResolvedEB.closure reb)
    case ebChecks? Γ la cert reb of λ where
      (no _)  → failure "the certified EB is not valid in the announcing block's environment"
      (yes (wf , refs , sz , tsz)) → success (ls₁ , CERTIFY-EB (la≡ , pend , d , e , vc , EB (wf , refs , sz , tsz , ext)))

  Computational-CERTIFY .completeness Γ ls (nothing , nothing) _ CERTIFY-None = refl
  Computational-CERTIFY .completeness Γ ls (just _ , nothing) _ ()
  Computational-CERTIFY .completeness Γ ls (nothing , just _) _ ()
  Computational-CERTIFY .completeness Γ ls (just cert , just reb) ls₁ (CERTIFY-EB {la = la} (la≡ , pend , d , e , vc , EB (wf , refs , sz , tsz , ext)))
    with pending? (CertifyEnv.lastApplied Γ)
  ... | no ¬pending = ⊥-elim (¬pending (la , la≡))
  ... | yes (la' , la≡')
    with trans (sym la≡') la≡
  ... | refl
    with pendingEB? Γ la cert reb
  ... | no ¬pend = ⊥-elim (¬pend pend)
  ... | yes _
    with delay? Γ la cert reb
  ... | no ¬d = ⊥-elim (¬d d)
  ... | yes _
    with sameEpoch? Γ la cert reb
  ... | no ¬e = ⊥-elim (¬e e)
  ... | yes _
    with validCert? Γ la cert reb
  ... | no ¬vc = ⊥-elim (¬vc vc)
  ... | yes _
    with recomputeProof ext | completeness _ _ _ _ ext
  ... | success (_ , ext') | refl
    with ebChecks? Γ la cert reb
  ... | no ¬c = ⊥-elim (¬c (wf , refs , sz , tsz))
  ... | yes _ = refl

      Computational-CHAIN : Computational _⊢_⇀⦇_,CHAIN⦈_ String
  Computational-CHAIN .computeProof Γ cs (b , reb) = do
    ls₁ , certStep ← computeProof (certifyEnv cs (blockSlot b)) (LedgerStateOf cs) (b .Block.ebCert , reb)
    nes , tickStep ← map₁ ⊥-elim $ computeProof {STS = _⊢_⇀⦇_,TICK⦈_} _ _ _
    (_ , _) , bbStep ← computeProof _ (LedgerStateOf nes , nes .NewEpochState.bcur) b
    success (_ , CHAIN (certStep , tickStep , bbStep))

  Computational-CHAIN .completeness _ cs (b , reb) _ (CHAIN {nes = nes} (certStep , tickStep , bbStep))
    with recomputeProof certStep | completeness _ _ _ _ certStep
  ... | success _ | refl
    with recomputeProof tickStep | completeness _ _ _ _ tickStep
  ... | success _ | refl
    with recomputeProof bbStep | completeness _ _ _ _ bbStep
  ... | success _ | refl = refl
