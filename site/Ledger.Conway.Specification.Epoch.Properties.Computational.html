module _ {eps : EpochState} {e : Epoch} where

  EPOCH-total : ∃[ eps' ] _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps'
  EPOCH-total =
    -, EPOCH
         ( SNAP-total      .proj₂
         , POOLREAP-total  .proj₂
         , RATIFIES-total' .proj₂)

  EPOCH-deterministic : ∀ eps' eps''
                      → _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps'
                      → _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps''
                      → eps' ≡ eps''
  EPOCH-deterministic
      eps'
      eps''
      (EPOCH
        {dState' = dState'₁}
        {acnt' = acnt'₁}
        {ss' = ss'₁}
        {utxoSt'' = utxoSt''₁}
        {pState'' = pState'₁}
        (p₁ , p₂ , p₃)
      )
      (EPOCH
        {dState' = dState'₂}
        {acnt' = acnt'₂}
        {ss' = ss'₂}
        {utxoSt'' = utxoSt''₂}
        {pState'' = pState'₂}
        (p₁' , p₂' , p₃')
      ) = eps'≡eps''
       where
         ls : LState
         ls = LStateOf eps

         es : EnactState
         es = EnactStateOf (RatifyStateOf eps)

         govUpd : Governance-Update
         govUpd = GovernanceUpdate.updates ls (RatifyStateOf eps)

         govSt' = Governance-Update.govSt' govUpd

         ss'₁≡ss'₂ : ss'₁ ≡ ss'₂
         ss'₁≡ss'₂ = SNAP-deterministic p₁ p₁'

         module pPRUpd =  Pre-POOLREAP-Update (Pre-POOLREAPUpdate.updates e ls es govUpd)

         pPRUpd₁ = Post-POOLREAPUpdate.updates es ls dState'₁ acnt'₁ govUpd
         module pPRUpd₁ = Post-POOLREAP-Update pPRUpd₁

         pPRUpd₂ = Post-POOLREAPUpdate.updates es ls dState'₂ acnt'₂ govUpd
         module pPRUpd₂ = Post-POOLREAP-Update pPRUpd₂

         prs'≡prs'' : $\begin{pmatrix} \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1769}{\htmlId{2786}{\htmlClass{Bound}{\text{utxoSt''₁}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1720}{\htmlId{2798}{\htmlClass{Bound}{\text{acnt'₁}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1693}{\htmlId{2807}{\htmlClass{Bound}{\text{dState'₁}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1800}{\htmlId{2818}{\htmlClass{Bound}{\text{pState'₁}}}}\, \end{pmatrix}$ ≡
                      $\begin{pmatrix} \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1949}{\htmlId{2856}{\htmlClass{Bound}{\text{utxoSt''₂}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1900}{\htmlId{2868}{\htmlClass{Bound}{\text{acnt'₂}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1873}{\htmlId{2877}{\htmlClass{Bound}{\text{dState'₂}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1980}{\htmlId{2888}{\htmlClass{Bound}{\text{pState'₂}}}}\, \end{pmatrix}$
         prs'≡prs'' = POOLREAP-deterministic-≡ refl refl p₂ p₂'

         pPRUpd₁≡pPRUpd₂ : pPRUpd₁ ≡ pPRUpd₂
         pPRUpd₁≡pPRUpd₂ =
           cong₂ (λ ds acnt → Post-POOLREAPUpdate.updates es ls ds acnt govUpd)
                 (cong PoolReapState.dState prs'≡prs'')
                 (cong PoolReapState.acnt prs'≡prs'')

         stakeDistrs₁≡stakeDistrs₂ : mkStakeDistrs (Snapshots.mark ss'₁) e pPRUpd.utxoSt' govSt' (GStateOf ls) pPRUpd₁.dState''
                                     ≡ mkStakeDistrs (Snapshots.mark ss'₂) e pPRUpd.utxoSt' govSt' (GStateOf ls) pPRUpd₂.dState''
         stakeDistrs₁≡stakeDistrs₂ =
           cong₂ (λ ss' upd → mkStakeDistrs (Snapshots.mark ss') e pPRUpd.utxoSt' govSt' (GStateOf ls) (Post-POOLREAP-Update.dState'' upd))
                 ss'₁≡ss'₂ pPRUpd₁≡pPRUpd₂

         Γ≡Γ' = cong₂ (λ sd acnt → $\begin{pmatrix} \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#3731}{\htmlId{3743}{\htmlClass{Bound}{\text{sd}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1238}{\htmlId{3748}{\htmlClass{Bound}{\text{e}}}}\, \\ \,\href{Ledger.Conway.Specification.Gov.Actions.html#16835}{\htmlId{3752}{\htmlClass{Field}{\text{DRepsOf}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#2059}{\htmlId{3760}{\htmlClass{Function}{\text{ls}}}}\, \\ \,\href{Ledger.Conway.Specification.Certs.html#1771}{\htmlId{3765}{\htmlClass{Field}{\text{CCHotKeysOf}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#2059}{\htmlId{3777}{\htmlClass{Function}{\text{ls}}}}\, \\ \,\href{Ledger.Prelude.Base.html#1263}{\htmlId{3782}{\htmlClass{Field}{\text{TreasuryOf}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#3734}{\htmlId{3793}{\htmlClass{Bound}{\text{acnt}}}}\, \\ \,\href{Ledger.Conway.Specification.Certs.html#1856}{\htmlId{3800}{\htmlClass{Field}{\text{PoolsOf}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#2059}{\htmlId{3808}{\htmlClass{Function}{\text{ls}}}}\, \\ \,\href{Ledger.Conway.Specification.Gov.Actions.html#8672}{\htmlId{3813}{\htmlClass{Field}{\text{VoteDelegsOf}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#2059}{\htmlId{3826}{\htmlClass{Function}{\text{ls}}}}\, \end{pmatrix}$)
                      stakeDistrs₁≡stakeDistrs₂ (cong Post-POOLREAP-Update.acnt'' pPRUpd₁≡pPRUpd₂)

         fut'≡fut'' : RatifyStateOf eps' ≡ RatifyStateOf eps''
         fut'≡fut'' = RATIFIES-deterministic-≡ Γ≡Γ' refl refl p₃ p₃'

         eps'≡eps'' : eps' ≡ eps''
         eps'≡eps'' =
           trans (cong₂ (λ ss prs → mkEpochState ss prs pPRUpd₁ (RatifyStateOf eps'))
                        ss'₁≡ss'₂ prs'≡prs'')
                 (cong₂ (mkEpochState ss'₂ $\begin{pmatrix} \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1949}{\htmlId{4299}{\htmlClass{Bound}{\text{utxoSt''₂}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1900}{\htmlId{4311}{\htmlClass{Bound}{\text{acnt'₂}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1873}{\htmlId{4320}{\htmlClass{Bound}{\text{dState'₂}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#1980}{\htmlId{4331}{\htmlClass{Bound}{\text{pState'₂}}}}\, \end{pmatrix}$)
                        pPRUpd₁≡pPRUpd₂ fut'≡fut'')
           where
             mkEpochState : Snapshots → PoolReapState → Post-POOLREAP-Update → RatifyState → EpochState
             mkEpochState ss prs upd fut =
               $\begin{pmatrix} \,\href{Ledger.Conway.Specification.Epoch.html#22865}{\htmlId{4577}{\htmlClass{Field}{\text{Post-POOLREAP-Update.acnt''}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#4550}{\htmlId{4605}{\htmlClass{Bound}{\text{upd}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#4543}{\htmlId{4611}{\htmlClass{Bound}{\text{ss}}}}\,
               \\ \begin{pmatrix} \,\href{Ledger.Conway.Specification.PoolReap.html#733}{\htmlId{4633}{\htmlClass{Field}{\text{PoolReapState.utxoSt}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#4546}{\htmlId{4654}{\htmlClass{Bound}{\text{prs}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#2284}{\htmlId{4660}{\htmlClass{Function}{\text{govSt'}}}}\,
                 \\ \begin{pmatrix} \,\href{Ledger.Conway.Specification.Epoch.html#22837}{\htmlId{4688}{\htmlClass{Field}{\text{Post-POOLREAP-Update.dState''}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#4550}{\htmlId{4718}{\htmlClass{Bound}{\text{upd}}}}\, \\ \,\href{Ledger.Conway.Specification.PoolReap.html#868}{\htmlId{4724}{\htmlClass{Field}{\text{PoolReapState.pState}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#4546}{\htmlId{4745}{\htmlClass{Bound}{\text{prs}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.html#21770}{\htmlId{4751}{\htmlClass{Function}{\text{pPRUpd.gState'}}}}\, \end{pmatrix} \end{pmatrix}
               \\ \,\htmlId{4789}{\htmlClass{Keyword}{\text{record}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#2108}{\htmlId{4796}{\htmlClass{Function}{\text{es}}}}\, \,\htmlId{4799}{\htmlClass{Symbol}{\text{{}}}\, \,\href{Ledger.Conway.Specification.Enact.html#1940}{\htmlId{4801}{\htmlClass{Field}{\text{withdrawals}}}}\, \,\htmlId{4813}{\htmlClass{Symbol}{\text{=}}}\, \,\href{Class.HasEmptySet.html#287}{\htmlId{4815}{\htmlClass{Field}{\text{∅}}}}\, \,\htmlId{4817}{\htmlClass{Symbol}{\text{}}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#4554}{\htmlId{4821}{\htmlClass{Bound}{\text{fut}}}}\, \end{pmatrix}$

  EPOCH-complete : ∀ eps' → _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps' → proj₁ EPOCH-total ≡ eps'
  EPOCH-complete eps' p = EPOCH-deterministic (proj₁ EPOCH-total) eps' (proj₂ EPOCH-total) p

  abstract
    EPOCH-total' : ∃[ eps' ] _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps'
    EPOCH-total' = EPOCH-total

    EPOCH-complete' : ∀ eps' → _ ⊢ eps ⇀⦇ e ,EPOCH⦈ eps' → proj₁ EPOCH-total' ≡ eps'
    EPOCH-complete' = EPOCH-complete

instance
  Computational-EPOCH : Computational _⊢_⇀⦇_,EPOCH⦈_ ⊥
  Computational-EPOCH .computeProof Γ s sig = success EPOCH-total'
  Computational-EPOCH .completeness Γ s sig s' h = cong success (EPOCH-complete' s' h)

module _ {e : Epoch} where

  NEWEPOCH-total : ∀ nes'' → ∃[ nes' ] _ ⊢ nes'' ⇀⦇ e ,NEWEPOCH⦈ nes'
  NEWEPOCH-total nes with e ≟ NewEpochState.lastEpoch nes + 1 | NewEpochState.ru nes | inspect NewEpochState.ru nes
  ... | yes p | just ru | PE.[ refl ] =  $\begin{pmatrix} \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#5453}{\htmlId{5700}{\htmlClass{Bound}{\text{e}}}}\, \\ \,\htmlId{5704}{\htmlClass{Symbol}{\text{\_}}}\, \\ \,\htmlId{5708}{\htmlClass{Symbol}{\text{\_}}}\, \\ \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#5018}{\htmlId{5712}{\htmlClass{Function}{\text{EPOCH-total'}}}}\, \,\htmlId{5725}{\htmlClass{Symbol}{\text{.}}}\,\,\href{Data.Product.Base.html#636}{\htmlId{5726}{\htmlClass{Field}{\text{proj₁}}}}\, \\ \,\href{Agda.Builtin.Maybe.html#194}{\htmlId{5734}{\htmlClass{InductiveConstructor}{\text{nothing}}}}\, \\ \,\htmlId{5744}{\htmlClass{Symbol}{\text{\_}}}\, \end{pmatrix}$
                                      , NEWEPOCH-New (p , EPOCH-total' .proj₂)
  ... | yes p | nothing | PE.[ refl ] = $\begin{pmatrix} \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#5453}{\htmlId{5869}{\htmlClass{Bound}{\text{e}}}}\, \\ \,\htmlId{5873}{\htmlClass{Symbol}{\text{\_}}}\, \\ \,\htmlId{5877}{\htmlClass{Symbol}{\text{\_}}}\, \\ \,\href{Data.Product.Base.html#636}{\htmlId{5881}{\htmlClass{Field}{\text{proj₁}}}}\, \,\href{Ledger.Conway.Specification.Epoch.Properties.Computational.html#5018}{\htmlId{5887}{\htmlClass{Function}{\text{EPOCH-total'}}}}\, \\ \,\href{Agda.Builtin.Maybe.html#194}{\htmlId{5902}{\htmlClass{InductiveConstructor}{\text{nothing}}}}\, \\ \,\htmlId{5912}{\htmlClass{Symbol}{\text{\_}}}\, \end{pmatrix}$
                                      , NEWEPOCH-No-Reward-Update (p , EPOCH-total' .proj₂)
  ... | no ¬p | _ | _ = -, NEWEPOCH-Not-New ¬p

  NEWEPOCH-complete : ∀ nes nes' → _ ⊢ nes ⇀⦇ e ,NEWEPOCH⦈ nes' → proj₁ (NEWEPOCH-total nes) ≡ nes'
  -- NEWEPOCH-complete nes nes' h with e ≟ NewEpochState.lastEpoch nes + 1 | NewEpochState.ru nes | h
  NEWEPOCH-complete nes nes' h with e ≟ NewEpochState.lastEpoch nes + 1 | NewEpochState.ru nes | inspect NewEpochState.ru nes | h
  ... | yes p | just ru | PE.[ refl ] | NEWEPOCH-New (x , x₁) rewrite EPOCH-complete' _ x₁ = refl
  ... | yes p | ru | PE.[ refl ] | NEWEPOCH-Not-New x = ⊥-elim $ x p
  ... | yes p | nothing | PE.[ refl ] | NEWEPOCH-No-Reward-Update (x , x₁) rewrite EPOCH-complete' _ x₁ = refl
  ... | no ¬p | ru | PE.[ refl ] | NEWEPOCH-New (x , x₁)  = ⊥-elim $ ¬p x
  ... | no ¬p | ru | PE.[ refl ] | NEWEPOCH-Not-New x = refl
  ... | no ¬p | nothing | PE.[ refl ] | NEWEPOCH-No-Reward-Update (x , x₁) = ⊥-elim $ ¬p x

instance
  Computational-NEWEPOCH : Computational _⊢_⇀⦇_,NEWEPOCH⦈_ ⊥
  Computational-NEWEPOCH .computeProof Γ s sig = success (NEWEPOCH-total _)
  Computational-NEWEPOCH .completeness Γ s sig s' h = cong success (NEWEPOCH-complete _ s' h)
